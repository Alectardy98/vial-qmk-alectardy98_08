cmake_minimum_required(VERSION 3.21)
project(qmk_ec_util VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
option(EC_ENABLE_SANITIZERS "Enable host address/undefined behavior checks" OFF)
if(EC_ENABLE_SANITIZERS)
 set(EC_HARNESS_SANITIZE 1)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
else()
 set(EC_HARNESS_SANITIZE 0)
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Test)
find_package(LibLZMA REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HIDAPI REQUIRED IMPORTED_TARGET hidapi-hidraw)
add_library(ec_core src/boardprofile.cpp src/layout.cpp src/protocol.cpp src/telemetry.cpp src/device.cpp src/settings.cpp src/calibration.cpp)
target_include_directories(ec_core PUBLIC src ../common/interfaces)
target_link_libraries(ec_core PUBLIC Qt6::Core Qt6::Gui PkgConfig::HIDAPI LibLZMA::LibLZMA)
target_compile_options(ec_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(ec_widgets src/widgets.cpp src/mainwindow.cpp src/livepanel.cpp)
target_link_libraries(ec_widgets PUBLIC ec_core Qt6::Widgets PkgConfig::HIDAPI)
target_compile_definitions(ec_widgets PRIVATE EC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
target_compile_options(ec_widgets PRIVATE -Wall -Wextra -Wpedantic)
add_executable(qmk-ec-util src/main.cpp)
target_link_libraries(qmk-ec-util PRIVATE ec_widgets)
target_compile_definitions(qmk-ec-util PRIVATE EC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
include(CTest)
get_filename_component(EC_DEFAULT_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(EC_FIRMWARE_DIR "${EC_DEFAULT_REPO_ROOT}/keyboards/alectardy98_ec/class0413_ec" CACHE PATH "Read-only Class0413 framework sources for host tests")
if(BUILD_TESTING)
 add_executable(ec_settings_tests tests/test_settings.cpp)
 target_link_libraries(ec_settings_tests PRIVATE ec_core Qt6::Test)
 add_test(NAME settings_safety COMMAND ec_settings_tests)
 set_tests_properties(settings_safety PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
 add_executable(ec_tests tests/test_core.cpp)
 target_link_libraries(ec_tests PRIVATE ec_widgets Qt6::Test)
 target_compile_definitions(ec_tests PRIVATE EC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
 add_test(NAME ec_core_and_ui COMMAND ec_tests)
 set_tests_properties(ec_core_and_ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
 add_test(NAME gui_smoke COMMAND qmk-ec-util --smoke-test)
 set_tests_properties(gui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
endif()
install(TARGETS qmk-ec-util RUNTIME DESTINATION bin)
install(DIRECTORY tests/fixtures/ DESTINATION share/qmk-ec-util/examples FILES_MATCHING PATTERN "*.json")

if(BUILD_TESTING)
 add_executable(ec_board_tests tests/test_boards.cpp)
 target_link_libraries(ec_board_tests PRIVATE ec_core Qt6::Test)
 target_compile_definitions(ec_board_tests PRIVATE EC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
 add_test(NAME universal_board_profiles COMMAND ec_board_tests)
 foreach(board class0413_ec ec_prox me_ec_uf2)
  add_test(NAME offline_${board} COMMAND qmk-ec-util --board ${board} --smoke-test)
  set_tests_properties(offline_${board} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
 endforeach()
endif()

if(BUILD_TESTING)
 get_filename_component(EC_REPO_ROOT "${EC_FIRMWARE_DIR}/../../.." ABSOLUTE)
 set(EC_FRAMEWORK_DIR "${EC_REPO_ROOT}/keyboards/alectardy98_ec" CACHE PATH "Framework firmware sources")
 # Use the interpreter selected by the build environment, as QMK make does.
 if(NOT Python3_EXECUTABLE)
  find_program(Python3_EXECUTABLE NAMES python3 REQUIRED)
 endif()
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 if(NOT EXISTS "${EC_REPO_ROOT}/development/alectardy98_ec/port-harness.py")
  message(FATAL_ERROR "Current EC42 source harness is required; restore development/alectardy98_ec/port-harness.py")
 endif()
 if(BUILD_TESTING)
  file(GLOB_RECURSE EC_PORT_SOURCES CONFIGURE_DEPENDS "${EC_FRAMEWORK_DIR}/common/*" "${EC_FRAMEWORK_DIR}/class0413_ec/*.c" "${EC_FRAMEWORK_DIR}/class0413_ec/*.h" "${EC_FRAMEWORK_DIR}/ec_prox/*.c" "${EC_FRAMEWORK_DIR}/ec_prox/*.h" "${EC_FRAMEWORK_DIR}/me_ec_uf2/*.c" "${EC_FRAMEWORK_DIR}/me_ec_uf2/*.h")
  file(GLOB EC_HARNESS_MOCKS CONFIGURE_DEPENDS "${EC_REPO_ROOT}/development/alectardy98_ec/tests/*-mocks.json")
  foreach(board class0413_ec ec_prox me_ec_uf2)
   add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${board}-harness"
    COMMAND ${CMAKE_COMMAND} -E env SANITIZE=${EC_HARNESS_SANITIZE} ${Python3_EXECUTABLE} "${EC_REPO_ROOT}/development/alectardy98_ec/port-harness.py" ${board} "${CMAKE_CURRENT_BINARY_DIR}/${board}-harness"
    DEPENDS ${EC_PORT_SOURCES} "${EC_REPO_ROOT}/development/alectardy98_ec/port-harness.py" ${EC_HARNESS_MOCKS})
   add_custom_target(${board}_harness DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/${board}-harness")
   add_test(NAME scanner_${board} COMMAND ${CMAKE_COMMAND} -E env ASAN_OPTIONS=detect_leaks=0 ${Python3_EXECUTABLE} "${EC_REPO_ROOT}/development/alectardy98_ec/tests/test_ports.py" ${board} "${CMAKE_CURRENT_BINARY_DIR}/${board}-harness")
   set_tests_properties(scanner_${board} PROPERTIES TIMEOUT 180)
  endforeach()
  add_executable(ec_live_tests tests/test_live.cpp)
  target_link_libraries(ec_live_tests PRIVATE ec_widgets Qt6::Test)
  target_compile_definitions(ec_live_tests PRIVATE EC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures" EC_FWHARNESS="${CMAKE_CURRENT_BINARY_DIR}/class0413_ec-harness")
  add_dependencies(ec_live_tests class0413_ec_harness)
  add_test(NAME current_live_protocol_offline COMMAND ec_live_tests)
  set_tests_properties(current_live_protocol_offline PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
  add_executable(ec_port_tests tests/test_ports_live.cpp)
  target_link_libraries(ec_port_tests PRIVATE ec_core Qt6::Test)
  target_compile_definitions(ec_port_tests PRIVATE EC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures" EC_CLASS_HARNESS="${CMAKE_CURRENT_BINARY_DIR}/class0413_ec-harness" EC_PROX_HARNESS="${CMAKE_CURRENT_BINARY_DIR}/ec_prox-harness" EC_MATRIX_HARNESS="${CMAKE_CURRENT_BINARY_DIR}/me_ec_uf2-harness")
  add_dependencies(ec_port_tests class0413_ec_harness ec_prox_harness me_ec_uf2_harness)
  add_test(NAME framework_port_live COMMAND ec_port_tests)
  set_tests_properties(framework_port_live PROPERTIES TIMEOUT 90)
 endif()
endif()
